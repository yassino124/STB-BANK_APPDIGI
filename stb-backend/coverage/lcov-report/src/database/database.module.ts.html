  private async seedDefaultData() {
    let existingRoles;
    try {
      existingRoles = await this.settingsService.findByKey('ROLES_SEEDED');
    } catch (e) {
      existingRoles = null;
    }
    if (existingRoles) return;
 
    // Seed default roles and permissions
